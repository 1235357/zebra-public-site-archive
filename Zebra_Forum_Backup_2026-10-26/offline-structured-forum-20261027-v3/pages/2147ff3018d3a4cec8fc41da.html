Valery, &nbsp;I think that the tag mask function is used to program a specific tag i.e. you first&nbsp;set the mask to match all the tagID bits as follows:
HANDLE hReader;TAG_MASK TagMask;

BYTE *pTagBitMask &nbsp;= ”\x10\x20\x30\x40\x50\x60\x70\x80”;

TagMask.cBitLen = 8*8; &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;// Match all 64 bits of the TAG IDTagMask.cBitStartOffset = 16; &nbsp;&nbsp;&nbsp;// Start matching after the first 16 bits.&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;// These bits are reserved by the tag.// set the tag mask bitsmemcpy(TagMask.cTagMask, pTagBitMask, 8);

RFID_SetTagMask(hReader, &amp;TagMask);Once you have set the mask , it should only program the TagID which is passed in the program function:

pTagID = "\x01\x02\x03\x04\x05\x06\x07\x08”;cTagLength = 8;cTagVerifyCount = 3;cEraseAttempts = 3;cProgramAttempts = 3;

dwStatus &nbsp;= RFID_ProgramTags(hRFIDReader, pTagID, cTagLength, cTagVerifyCount, cEraseAttempts, cProgramAttempts);if(dwStatus == RFID_SUCCESS)

{&nbsp;&nbsp;// Tag programmed};&nbsp;The above code extracts are from the EMDK for C help file which contains some useful info+code snippets&nbsp;on the API.Best Regards,Ian HattonEMb EMEA
