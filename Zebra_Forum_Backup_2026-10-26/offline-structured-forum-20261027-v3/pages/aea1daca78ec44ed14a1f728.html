Are you using&nbsp;Link-OS SDK for PC-.NET or for PC-Java? The syntaxes are slightly different.

// For .NET
string theIpAddress="10.10.1.150";
Connection connection = ConnectionBuilder.Build($"TCP:{theIpAddress}:9100");

// For Java
String theIpAddress="10.10.1.150";
Connection connection = ConnectionBuilder.build("TCP:" + theIpAddress + ":9100");
