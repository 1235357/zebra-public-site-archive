Issue Summary

I'm experiencing a persistent "Operation In Progress" error when attempting to write to UHF RFID tags using the Zebra Android SDK. Despite implementing multiple conflict prevention strategies, the error continues to occur intermittently, making reliable tag writing impossible.

Hardware/Software Details

  • Device: RFD90
  • SDK Version: 2.0.5.224
  • Tag Type: UHF Gen2 tags (various manufacturers - Impinj, NXP)

Current Implementation
private suspend fun performWrite(
   tagId: String,
   memoryBank: MEMORY_BANK,
   offset: Int,
   data: String,
   accessPassword: Long,
   operationType: String
): Boolean {
   return withContext(Dispatchers.IO) {
       try {
           Log.d(TAG, "Starting atomic write for $operationType")
           val config = reader!!.Config.Antennas.getAntennaRfConfig(1)
           config.transmitPowerIndex = 240
           config.setrfModeTableIndex(0)
           config.tari = 0
           reader!!.Config.Antennas.setAntennaRfConfig(1, config)

           reader!!.Config.setAccessOperationWaitTimeout(15000)
           reader!!.Actions.Inventory.stop()

           val tagAccess = TagAccess()
           val accessParams = tagAccess.WriteAccessParams()
           accessParams.accessPassword = accessPassword
           accessParams.memoryBank = memoryBank
           accessParams.offset = offset
           accessParams.setWriteData(data)
           accessParams.writeRetries = 3
           accessParams.writeDataLength = data.length / 4

           Log.d(TAG, "Executing write with TID-based password: ${accessPassword.toString(16).uppercase()}")

           // wait 150ms before write
           kotlinx.coroutines.delay(150)
           val useTIDfilter = memoryBank === MEMORY_BANK.MEMORY_BANK_EPC
           reader!!.Actions.TagAccess.writeWait(tagId, accessParams, null, null, true, useTIDfilter)

           Log.d(TAG, "Write for $operationType completed successfully")
           return@withContext true

       } catch (e: OperationFailureException) {
           Log.e(TAG, "Write failed for $operationType: ${e.vendorMessage}")
           throw e
       }
   }
}
