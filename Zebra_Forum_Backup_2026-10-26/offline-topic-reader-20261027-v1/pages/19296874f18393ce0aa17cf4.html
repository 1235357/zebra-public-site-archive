                  Hello,I am currently trying to automatically upload a dataWedge profile from my android mobile application to the dataWedge application via the Intent IMPORT_CONFIG My profile is in the folder raw inside the application and I copy it to the external directory available for my application (sdcard/data/com.myapplication.app/files). I enable logging for dataWedge and don't have any error when the profile is send and can't be seen inside the  datawedge application. Since Android restricted writing in external storage, i can't upload the file elsewhere (except if I missed something, I tried with asking READ_EXTERNAL_STORAGE and WRITE_EXTERNAL_STORAGE permission at runtime but nothing change , same with MEDIA permissions)  private fun autoImportDataWedgeProfile() {

        val outFile = File(getExternalFilesDir(null), "dwprofile_profilename.db")
        outFile.setWritable(true, false)
        outFile.setExecutable(true, false)
        outFile.setReadable(true, false)

        resources.openRawResource(R.raw.dwprofile_profilename).use { input ->
            FileOutputStream(outFile).use { output ->
                exportFile(input, output)
            }
        }

        Log.e("DataWedge", "Profile copied successfully")
        importConfig()
    }

 private fun importConfig() {
        val bMain = Bundle()
        bMain.putString("FOLDER_PATH", getExternalFilesDir(null)?.path)
        val fileNames = ArrayList<String>()

        fileNames.add("dwprofile_profilename.db")

        bMain.putStringArrayList("FILE_LIST", fileNames)

        // send the intent
        val i = Intent()
        i.setAction("com.symbol.datawedge.api.ACTION")
        i.putExtra("com.symbol.datawedge.api.IMPORT_CONFIG", bMain)

        this.sendBroadcast(i)
    } Maybe I missed something in the documentation but did anyone tried with the intent and have you succeed ?  Thanks in advance for your help
