            Adding these 2 lines (minifyEnable, shrinkResources) to the Module's build.gradle file seems to have fixed the issue for me
buildTypes {
        release {
            minifyEnabled false
            shrinkResources false
            signingConfig signingConfigs.debug
        }
    }
