            I think you are mixing things and not getting the right result.You can use Inventory with interval =0 seconds which means the reader will be reporting tags all the time. Select tagmetadata to contain ANTENNA so that you get all reads with the antenna info. When you use the Inventory mode report filter is normally not effective and in latest version should not be available in the web interface.Next best thing is to use Custom where you can specify antenna start and stop conditions and probably specify the dwell time per antenna.here is an example: { "type": "SIMPLE", "modeSpecificSettings": {   "interval": {     "unit": "hours",     "value": -63127613   } }, "environnement":  "antennas": [   -50521876,   51243269 ], "filter": {   "value": "[a-zA-Z0-9]{2,}",   "match": "regex",   "operation": "include" }, "transmitPower": -70556978, "linkProfile": 12340601, "antennaStopCondition": {   "type": "SINGLE_INVENTORY_LIMITED_DURATION",   "value": -19319971.90736863 }, "query": {   "tagPopulation": 61863464,   "sel": "SL",   "session": "S3",   "target": "B" }, "tagMetaData": [   "XPC",   "CHANNEL" ], "radioStartConditions": {   "type": "GPI_WITH_RESTART",   "gpis": [     {       "port": 1,       "signal": "HIGH",       "debounceTime": 0     },     {       "port": 1,       "signal": "HIGH",       "debounceTime": 0     }   ] }, "radioStopConditions": {   "duration": -91418143,   "antennaCycles": 35293144,   "tagCount": 76942500,   "durationAfterNoMoreUniqueTags": 86348421,   "gpis": [     {       "port": 1,       "signal": "HIGH",       "debounceTime": 0     },     {       "port": 1,       "signal": "HIGH",       "debounceTime": 0     }   ] }, "reportFilter": {   "duration": 15261567,   "type": "RADIO_WIDE" }, "rssiFilter": {   "threshold": -22 }}
